# Opens settings with F1, selects Throttle's first key slot with A (waits for a
# key), cancels with B, then closes the menu with F1.
# Run: INPUT=tests/headless/settings_menu.txt tools/headless_run.sh 40 --vulkan_device=1 --vulkan_headless_surface=true
wait 24
focus trials
tap F1
wait 1.5
capture settings_open
pad down 1
wait 0.15
pad down 0
wait 0.5
pad a 1
wait 0.15
pad a 0
wait 1
capture settings_waiting
pad b 1
wait 0.15
pad b 0
wait 1
capture settings_cancelled
pad b 1
wait 0.15
pad b 0
wait 1.5
tap F1
wait 1.5
capture settings_closed
